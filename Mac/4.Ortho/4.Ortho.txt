@implementation MyOpenGLView
{
	@private

	CVDisplayLinkRef displayLink;

	//Paste all vao,vbo...all global variables here
}

-(id)initWithFrame:(NSRect)Frame
{
	//NO CHANGE
	self = [super initWithFrame:Frame];
	if(self)
	{
		NSOpenGLPixelFormatAttribute attributes[] = 
													{
														NSOpenGLPFAOpenGLProfile,NSOpenGLProfileVersion4_1Core,
														NSOpenGLPFAScreenMask,CGDisplayIDToOpenGLDisplayMask(kCGDirectMainDisplay);/*Take monitor to which gtaphics card connected*/  
														NSOpenGLPFANoRecovery,/*Don't Shift To S/W rendering*/
														NSOpenGLPFAAccelerated,/*Want only HardWare Rendering*/
														NSOpenGLPFAColorSize,24,
														NSOpenGLPFADepthSize,24,
														NSOpenGLPFAAlphaSize,8,
														NSOpenGLPFADoubleBuffer,0
													};  //This resembles with PFD in Widnows

		NSOpenGLPixelFormat *pixelFormat = [[[NSopenGLPixelFormate alloc]initWithAttributes:attributes]autorelease];
		
		if(pixelFormat == nil)
		{
			fprintf(gpFile,"Can't get Pixel Format...\n");
			[self release];
			[NSApp terminate:self];
		}

		NSOpenGLContext *glContext = [[[NSOpenGLContext alloc]initWithFormat:pixelFormat shareContext:nil]autorelease];
		
		[self setPixelFormat:pixelFormat]; // SetPixelFormat();

		[self setOpenGLContext:glContext];

	}
	return(self);
}

-(CVReturn)getFrameForTime:(const CVTimeStamp*)outputTime
{
	//NO CHANGE
	//code
	NSAutoreleasePool *pool = [[NSAutoreleasePool alloc]init]; //Apple's OpenGL supports multithreading..."Main" has main thread...This thread is for rendering
	[self drawView];
	[pool release];
	return(kCVReturnSuccess);
}

-(void)prepareOpenGL  //Like Initialize()
{	
	//WILL CHANGED
	//code
	[super prepareOpenGL];

	//fprintf(gpFile,"OpenGL Version"); 
	//fprintf(gpFile,"GLSL Version");

	[[self openGLContext]makeCurrentContext]; // In Objective C : getContext = direct Write context name ; setContext = "set" should be there..

	//Swap Interval
	GLint swapInt = 1;
	[[self openGLContext]setValue:&swapInt forParameter:NSOpenGLCPSwapInterval];//CP = Context Parameter

	//const char *vertexShaderSourceCode...All Code Copy 
	//upto glBindVertexArray(0)....All code cpy paste upto here...

	//HERE VERSION MUST BE "#vesrion 410 core"
	//Uninitialze => [self release]
	//exit(0) => [NSApp terminate:self]

	//3 Romantic Lines

	glClearColor(0.0f,0.0f,1.0f,1.0f);

	//CV and CGL related Code
	CVDisplayLinkCreateWithActiveCGDisplays(&displayLink);
	CVDisplayLinkSetOutputCallback(displayLink,&MyDisplayLinkCallback,self);
	CGLContextObj cglContext = (CGLContextObj)[[self openGLContext]CGLContextObj];
	CGLPixelFormatObj cglPixelFormat = (CGLPixelFormatObj)[self pixelFormat];
	CVDisplayLinkSetCurrentCGDisplayFromOpeGLContext(displayLink,cglContext,cglPixelFormat);
	CVDisplayLinkStart(displayLink);
}

-(void)reshape
{	
	//WILL CHANGED
	//code
	[super reshape];
	CGLLockContext((CGLContextObj)[[self openGLContext]CGLContextObj]);

	NSRect rect = [self bounce];
	if(rect.size.height < 0)
	{
		rect.size.height = 1;
	}
	glViewport(0,0,(GLsizei)rect.size.width,(GLsizei)rect.size.height);
	CGLUnlockContext((CGLContextObj)[[self openGLContext]CGLContextObj]);
}

-(void)drawRect:(NSRect)dirtyRect
{
	//code
	[self drawView];
}

-(void)drawView
{
	//WILL CHANGED
	//code
	[[self openGLContext]makeCurrentContext];
	CGLLockContext((CGLContextObj)[[self openGLContext]CGLContextObj]);
	glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

	//All code from Window Same...

	CGLFlushDrawable((CGLContextObj)[[self openGLContext]CGLContextObj]);
	CGLUnlockContext((CGLContextObj)[[self openGLContext]CGLContextObj]);
}

//Global 
CVReturn MyDisplayLinkCallback(CVDisplayLinkRef displayLink,const CVTimeStamp* now,const CVTimeStamp* outputTime,CVOptionFlags flagsIn,CVOptionFlags* flagsOut,void* displayLinkContext)
{	
	CVReturn result = [(myOpenGLView *)displayLinkContext getFrameForTime:outputTime]; //Here last param, void *displayLinkContext is nothing but myOpenGLView pointer set by OS...
	return(result);
}
